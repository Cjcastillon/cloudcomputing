<script>
/* ===== CONFIG ===== */
const OWNER = "Cjcastillon";
const REPO = "cloudcomputing";
const BRANCH = "Public";
/* ================== */

async function uploadFile() {
    const token = document.getElementById("tokenInput").value;
    const fileInput = document.getElementById("fileInput");
    const status = document.getElementById("statusText");

    if (!token) {
        status.textContent = "TOKEN REQUIRED";
        return;
    }
    if (!fileInput.files.length) {
        status.textContent = "NO FILE SELECTED";
        return;
    }

    const file = fileInput.files[0];
    status.textContent = "CHECKING FILE...";

    const base64 = await new Promise(resolve => {
        const reader = new FileReader();
        reader.onload = () => resolve(reader.result.split(",")[1]);
        reader.readAsDataURL(file);
    });

    const url = `https://api.github.com/repos/${OWNER}/${REPO}/contents/${file.name}`;
    let sha = null;

    const check = await fetch(url, {
        headers: {
            "Authorization": `Bearer ${token}`,
            "Accept": "application/vnd.github.v3+json"
        }
    });

    if (check.ok) {
        const data = await check.json();
        sha = data.sha;
        status.textContent = "OVERWRITING FILE...";
    } else {
        status.textContent = "UPLOADING FILE...";
    }

    const body = {
        message: sha ? `Update ${file.name}` : `Upload ${file.name}`,
        content: base64,
        branch: BRANCH,
        ...(sha && { sha })
    };

    const response = await fetch(url, {
        method: "PUT",
        headers: {
            "Authorization": `Bearer ${token}`,
            "Content-Type": "application/json",
            "Accept": "application/vnd.github.v3+json"
        },
        body: JSON.stringify(body)
    });

    const result = await response.json();

    status.textContent = response.ok
        ? (sha ? "FILE OVERWRITTEN ✔" : "UPLOAD SUCCESS ✔")
        : "ERROR: " + result.message;
}

async function fetchFile() {
    const token = document.getElementById("tokenInput").value;
    const fileName = document.getElementById("fileName").value;
    const img = document.getElementById("imagePreview");
    const text = document.getElementById("textPreview");
    const status = document.getElementById("statusText");

    if (!fileName) {
        status.textContent = "ENTER FILE NAME";
        return;
    }

    img.style.display = "none";
    text.style.display = "none";
    status.textContent = "FETCHING FILE...";

    const url = `https://api.github.com/repos/${OWNER}/${REPO}/contents/${fileName}?ref=${BRANCH}`;

    try {
        const response = await fetch(url, {
            headers: {
                "Authorization": `Bearer ${token}`,
                "Accept": "application/vnd.github.v3+json"
            }
        });

        if (!response.ok) throw new Error();

        const data = await response.json();

        if (data.name.match(/\.(jpg|jpeg|png|gif|webp|jfif)$/i)) {
            img.src = `data:image/*;base64,${data.content}`;
            img.style.display = "block";
        } else {
            text.value = atob(data.content);
            text.style.display = "block";
        }

        status.textContent = "FETCH SUCCESS ✔";
    } catch {
        status.textContent = "FETCH FAILED ✖";
    }
}
